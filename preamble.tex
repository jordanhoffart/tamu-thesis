% The algorithm environment needs LaTeX 2026-11 or newer to work with tagging.
% On older LaTeX (e.g. Overleaf's static TeX Live 2026 image), it fails to compile,
% so this swaps in a non-floating version that looks the same.
% On a recent enough LaTeX this does nothing; you can delete it once you're on one.
\NewDocumentEnvironment{plainalgorithm}{o}{
  \par\addvspace{\intextsep}
  \begin{singlespace}
  \RenewDocumentCommand{\caption}{m}{
    \refstepcounter{algorithm}
    \hrule height 0.8pt \kern 2pt
    \noindent\textbf{Algorithm \thealgorithm} ##1\par
    \kern 2pt \hrule \kern 2pt
  }
}{
  \par \kern 2pt \hrule
  \end{singlespace}
  \addvspace{\intextsep}
}
